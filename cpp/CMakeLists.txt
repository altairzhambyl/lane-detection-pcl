cmake_minimum_required(VERSION 3.8)
project(ransac_seg_cpp)

# The Python package is the reference implementation; this one exists to carry
# the per-sweep hot path, so it is built for speed and warned at hard.
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -O2)

find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)

# Deliberately NOT PCL, and please do not add it. The two things it would have
# been reached for are the exact Euclidean distance transform — 80 lines of
# Felzenszwalb & Huttenlocher in src/edt.cpp, pinned against a brute-force
# reference with exact double equality — and the neighbour searches in
# cluster/lane, which run over at most a few thousand vertices, where a uniform
# bucket grid measured faster than a kd-tree and is deterministic. PCL would buy
# nothing measurable here and would cost every downstream user a heavy
# dependency. Adding it needs a measurement, not a preference.

# --- ransac_seg_core: every algorithm, no ROS -------------------------------
# ROS-free on purpose, so bench_ground, bench_pipeline and the eleven algorithm
# gtests all build on a bare checkout with no sourced ROS installation.
add_library(ransac_seg_core
  src/ground_plane.cpp
  src/cost_grid.cpp
  src/params.cpp
  src/plane_filter.cpp
  src/numeric.cpp
  src/costmap.cpp
  src/edt.cpp
  src/planners.cpp
  src/cluster.cpp
  src/lane.cpp
  src/spline.cpp)
target_include_directories(ransac_seg_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(ransac_seg_core PUBLIC Eigen3::Eigen)

# Optional. lane.cpp guards its only pragma with #ifdef _OPENMP and its outputs
# were verified bit-identical with and without it, so a toolchain without
# OpenMP builds the same package, just slower on the lane projection loop.
find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
  target_link_libraries(ransac_seg_core PUBLIC OpenMP::OpenMP_CXX)
endif()

# --- ransac_seg_ros: the node layer -----------------------------------------
add_library(ransac_seg_ros
  src/cloud_io.cpp
  src/markers.cpp
  src/field_markers.cpp
  src/ros_params.cpp
  src/perception_core.cpp
  src/consumers.cpp
  src/ground_segmentation.cpp
  src/lane_boundary.cpp
  src/astar_planner.cpp
  src/dijkstra_planner.cpp
  src/bfs_planner.cpp
  src/dfs_baseline.cpp)
target_link_libraries(ransac_seg_ros PUBLIC ransac_seg_core)
ament_target_dependencies(ransac_seg_ros PUBLIC
  rclcpp rcl_interfaces sensor_msgs visualization_msgs geometry_msgs builtin_interfaces)

# --- executables ------------------------------------------------------------
# One decode/segment/rasterise per sweep feeding all six consumers. This is the
# launch default and the point of the whole exercise.
add_executable(perception_stack src/perception_stack_main.cpp)
target_link_libraries(perception_stack ransac_seg_ros)

# The same hub carrying ONE consumer, chosen by the `consumer` parameter, so the
# six-process Python topology stays available for A/B measurement.
add_executable(perception_node src/single_consumer_main.cpp)
target_link_libraries(perception_node ransac_seg_ros)

# Scores the port against SemanticKITTI labels, printing the same quantities as
# tools/eval_semantickitti.py so the two are comparable rather than both merely
# "fast". Needs no ROS, so it runs on a bare checkout.
add_executable(bench_ground src/bench_ground.cpp)
target_link_libraries(bench_ground ransac_seg_core)

# Per-stage p50/p95/p99 over real sweeps, plus the shared-hub-versus-six-copies
# measurement the architectural claim rests on.
add_executable(bench_pipeline src/bench_pipeline.cpp)
target_link_libraries(bench_pipeline ransac_seg_core)

install(TARGETS ransac_seg_core ransac_seg_ros
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin)
install(TARGETS perception_stack perception_node bench_ground bench_pipeline
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)

# launch/algorithms_cpp.launch.py sits beside the Python package's launch files
# at the repository root and is installed by setup.py's existing glob, so it is
# deliberately NOT installed twice from here.

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)

  # test_lane and test_spline replay golden dumps generated by
  # tools/dump_golden.py; without this path they fall back to "test/data"
  # relative to the working directory and report a regeneration hint.
  set(TEST_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/test/data")

  foreach(name
      ground_plane cost_grid params plane_filter numeric
      edt costmap planners cluster spline lane)
    ament_add_gtest(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} ransac_seg_core)
    target_compile_definitions(test_${name}
      PRIVATE RANSAC_SEG_CPP_TEST_DATA="${TEST_DATA_DIR}")
  endforeach()

  foreach(name cloud_io markers ros_params perception_hub)
    ament_add_gtest(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} ransac_seg_ros)
  endforeach()
endif()

ament_package()
